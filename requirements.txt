# DATABASE

def get_db():
    con = sqlite3.connect(
        DATABASE,
        timeout=30,
        check_same_thread=False
    )
    con.row_factory = sqlite3.Row
    return con


def init_db():

    with db_lock:

        con = get_db()

        cur = con.cursor()

        cur.execute("""
        CREATE TABLE IF NOT EXISTS users(
            user_id INTEGER PRIMARY KEY,
            username TEXT DEFAULT '',
            first_name TEXT DEFAULT '',
            balance REAL DEFAULT 0,
            created_at INTEGER,
            updated_at INTEGER
        )
        """)

        cur.execute("""
        CREATE TABLE IF NOT EXISTS payments(
            id INTEGER PRIMARY KEY AUTOINCREMENT,
            user_id INTEGER,
            kind TEXT,
            method TEXT,
            amount REAL,
            transaction_reference TEXT DEFAULT '',
            screenshot_file_id TEXT DEFAULT '',
            account_number TEXT DEFAULT '',
            account_name TEXT DEFAULT '',
            status TEXT DEFAULT 'PENDING',
            created_at INTEGER,
            reviewed_at INTEGER,
            reviewer_id INTEGER
        )
        """)

        cur.execute("""
        CREATE TABLE IF NOT EXISTS wallet_transactions(
            id INTEGER PRIMARY KEY AUTOINCREMENT,
            user_id INTEGER,
            type TEXT,
            amount REAL,
            balance_before REAL,
            balance_after REAL,
            reference TEXT,
            created_at INTEGER
        )
        """)

        cur.execute("""
        CREATE TABLE IF NOT EXISTS bets(
            id INTEGER PRIMARY KEY AUTOINCREMENT,
            user_id INTEGER,
            game TEXT,
            amount REAL,
            reference TEXT,
            status TEXT,
            created_at INTEGER
        )
        """)

        con.commit()
        con.close()

def ensure_user(user_id, username="", first_name=""):

    with db_lock:

        con = get_db()
        cur = con.cursor()

        row = cur.execute(
            "SELECT user_id FROM users WHERE user_id=?",
            (user_id,)
        ).fetchone()

        if row is None:

            cur.execute("""
            INSERT INTO users(
                user_id,
                username,
                first_name,
                balance,
                created_at,
                updated_at
            )
            VALUES(?,?,?,?,?,?)
            """, (
                user_id,
                username,
                first_name,
                0,
                int(time.time()),
                int(time.time())
            ))

        else:

            cur.execute("""
            UPDATE users
            SET username=?,
                first_name=?,
                updated_at=?
            WHERE user_id=?
            """, (
                username,
                first_name,
                int(time.time()),
                user_id
            ))

        con.commit()
        con.close()


def get_balance(user_id):

    with db_lock:

        con = get_db()

        row = con.execute(
            "SELECT balance FROM users WHERE user_id=?",
            (user_id,)
        ).fetchone()

        con.close()

    if row:
        return float(row["balance"])

    return 0

def change_balance(
    user_id,
    amount,
    transaction_type,
    reference=""
):

    with db_lock:

        con = get_db()
        cur = con.cursor()

        row = cur.execute(
            "SELECT balance FROM users WHERE user_id=?",
            (user_id,)
        ).fetchone()

        if not row:

            con.close()
            return False, 0

        before = float(row["balance"])
        after = before + amount

        if after < 0:

            con.close()
            return False, before

        cur.execute("""
        UPDATE users
        SET balance=?,
            updated_at=?
        WHERE user_id=?
        """, (
            after,
            int(time.time()),
            user_id
        ))

        cur.execute("""
        INSERT INTO wallet_transactions(
            user_id,
            type,
            amount,
            balance_before,
            balance_after,
            reference,
            created_at
        )
        VALUES(?,?,?,?,?,?,?)
        """, (
            user_id,
            transaction_type,
            amount,
            before,
            after,
            reference,
            int(time.time())
        ))

        con.commit()
        con.close()

    return True, after

def create_payment(
    user_id,

            user_id,

            transaction_reference,
            screenshot_file_id,
            account_number,
            account_name,
            status,
            created_at
        )
        VALUES(?,?,?,?,?,?,?,?,?,?)
        """, (
            user_id,
            kind,
            method,
            amount,
            reference,
            screenshot,
            account_number,
            account_name,
            "PENDING",
            int(time.time())
        ))

        payment_id = cur.lastrowid

        con.commit()
        con.close()

    return payment_id

async def text_handler(
    update: Update,
    context: ContextTypes.DEFAULT_TYPE
):

    text = update.message.text.strip()

    # ---------- CUSTOM DEPOSIT ----------

    if context.user_data.get(
        "waiting_deposit_amount"
    ):

        try:
            amount = float(text)
        except:
            await update.message.reply_text(
                "❌ ትክክለኛ የገንዘብ መጠን ያስገቡ።"
            )
            return

        if amount < MIN_DEPOSIT:

            await update.message.reply_text(
                f"❌ Minimum deposit is "
                f"{MIN_DEPOSIT} ETB."
            )
            return

        context.user_data.pop(
            "waiting_deposit_amount",
            None
        )

        context.user_data[
            "deposit_amount"
        ] = amount

        method = context.user_data.get(
            "deposit_method"
        )

        if method == "telebirr":

            account = (
                f"📱 Telebirr: {TELEBIRR_NUMBER}\n"
                f"👤 {TELEBIRR_NAME}"
            )

        else:

            account = (
                f"🏦 CBE: {CBE_ACCOUNT}\n"
                f"👤 {CBE_NAME}"
            )

        keyboard = InlineKeyboardMarkup([

            [
                InlineKeyboardButton(
                    "✅ I PAID",
                    callback_data="dep_paid"
                )
            ],

            [
                InlineKeyboardButton(
                    "❌ Cancel",
                    callback_data="home"
                )
            ]
        ])

        await update.message.reply_text(
            "💳 CONFIRM DEPOSIT\n\n"
            f"Method: {method.upper()}\n"
            f"Amount: {amount} ETB\n\n"
            f"{account}\n\n"
            "ክፍያውን ካደረጉ I PAID ይጫኑ።",
            reply_markup=keyboard
        )

        return

    # ---------- CUSTOM WITHDRAW ----------

    if context.user_data.get(
        "waiting_withdraw_amount"
    ):

        try:
            amount = float(text)
        except:
            await update.message.reply_text(
                "❌ ትክክለኛ መጠን ያስገቡ።"
            )
            return

        balance = get_balance(
            update.effective_user.id
        )

        if amount < MIN_WITHDRAW:

            await update.message.reply_text(
                f"❌ Minimum withdrawal is "
                f"{MIN_WITHDRAW} ETB."
            )
            return

        if amount > balance:

            await update.message.reply_text(
                f"❌ Insufficient balance.\n"
                f"Available: {balance:.2f} ETB"
            )
            return

        context.user_data.pop(
            "waiting_withdraw_amount",
            None
        )

        context.user_data[
            "withdraw_amount"
        ] = amount

        context.user_data[
            "waiting_withdraw_account"
        ] = True

        await update.message.reply_text(
            "📱 TELEBIRR WITHDRAWAL\n\n"
            f"Amount: {amount} ETB\n\n"
            "የTelebirr ቁጥር ያስገቡ።"
        )

        return

    # ---------- WITHDRAW ACCOUNT ----------

    if context.user_data.get(
        "waiting_withdraw_account"
    ):

        if not text.isdigit() or not (
            9 <= len(text) <= 20
        ):

            await update.message.reply_text(
                "❌ ትክክለኛ Telebirr "
                "account number ያስገቡ።"
            )

        context.user_data[
            "withdraw_account"
        ] = text

        context.user_data.pop(
            "waiting_withdraw_account",
            None
        )

        context.user_data[
            "waiting_withdraw_name"
        ] = True

        await update.message.reply_text(
            "👤 የTelebirr account ባለቤት "
            "ስም ያስገቡ።"
        )

        return

    # ---------- WITHDRAW NAME ----------

    if context.user_data.get(
        "waiting_withdraw_name"
    ):

        context.user_data[
            "withdraw_name"
        ] = text

        context.user_data.pop(
            "waiting_withdraw_name",
            None
        )

        await show_withdraw_confirm(
            update,
            context
        )

        return

    # ---------- TRANSACTION NUMBER ----------

    if context.user_data.get(
        "waiting_transaction"
    ):

        reference = text

        if len(reference) < 3:

            await update.message.reply_text(
                "❌ Transaction number ያስገቡ።"
            )
            return

        photo = context.user_data.get(
            "deposit_photo"
        )

        method = context.user_data.get(
            "deposit_method"
        )

        amount = context.user_data.get(

# ============================================================
# ADMIN APPROVE / REJECT
# ============================================================

async def admin_review(
    query,
    payment_id,
    approve
):

    if query.from_user.id != ADMIN_ID:

        await query.answer(
            "❌ Admin only.",
            show_alert=True
        )
        return

    payment = get_payment(payment_id)

    if not payment:

        await query.answer(
            "Payment not found.",
            show_alert=True
        )
        return

    if payment["status"] != "PENDING":

        await query.answer(
            "Already reviewed.",
            show_alert=True
        )
        return

    new_status = (
        "APPROVED"
        if approve
        else "REJECTED"
    )

    # ================= DEPOSIT =================

    if payment["kind"] == "DEPOSIT":

        if approve:

            success, balance = change_balance(
                payment["user_id"],
                payment["amount"],
                "DEPOSIT",
                f"payment:{payment_id}"
            )

            if not success:

                await query.answer(
                    "Balance update failed.",
                    show_alert=True
                )
                return

        else:

            balance = get_balance(
                payment["user_id"]
            )

    # ================= WITHDRAW =================

    else:

        if approve:

            success, balance = change_balance(
                payment["user_id"],
                -payment["amount"],
                "WITHDRAW",
                f"payment:{payment_id}"
            )

            if not success:

                await query.answer(
                    "Insufficient balance.",
                    show_alert=True
                )
                return

        else:

            balance = get_balance(
                payment["user_id"]
            )

    with db_lock:

        con = get_db()

        con.execute("""
        UPDATE payments
        SET status=?,
            reviewed_at=?,
            reviewer_id=?
        WHERE id=?
        """, (
            new_status,
            int(time.time()),
            ADMIN_ID,
            payment_id
        ))

        con.commit()
        con.close()

    await query.edit_message_text(
        "✅ APPROVED" if approve
        else "❌ REJECTED"
    )

    # Notify user

    if payment["kind"] == "DEPOSIT":

        if approve:

            message = (
                "✅ DEPOSIT APPROVED\n\n"
                f"Amount: {payment['amount']:.2f} ETB\n"
                f"💰 New balance: {balance:.2f} ETB"
            )

        else:

            message = (
                "❌ DEPOSIT REJECTED\n\n"
                "የDeposit requestዎ ተቀባይነት አላገኘም።"
            )

    else:

        if approve:

            message = (
                "✅ WITHDRAWAL APPROVED\n\n"
                f"Amount: {payment['amount']:.2f} ETB\n"
                f"💰 New balance: {balance:.2f} ETB"
            )

        else:

            message = (
                "❌ WITHDRAWAL REJECTED\n\n"
                "የWithdrawal requestዎ ተቀባይነት አላገኘም።"
            )

    try:

        await query.get_bot().send_message(
            chat_id=payment["user_id"],
            text=message,
            reply_markup=home_keyboard()
        )

    except Exception as e:

        print("User notification error:", e)

# CALLBACKS
# ============================================================

async def callback(
    update: Update,
    context: ContextTypes.DEFAULT_TYPE
):

    query = update.callback_query

    await query.answer()

    data = query.data

    # HOME

    if data == "home":

        context.user_data.clear()

        await query.edit_message_text(
            "👑 KING BINGO\n"
            "━━━━━━━━━━━━━━━━\n\n"
            "🎮 Play King Bingo\n"
            "💰 Wallet\n"
            "➕ Deposit\n"
            "➖ Withdraw",
            reply_markup=home_keyboard()
        )

        return

    # WALLET

    if data == "wallet":

        balance = get_balance(
            query.from_user.id
        )

        await query.edit_message_text(
            "💰 WALLET\n"
            "━━━━━━━━━━━━━━━━\n\n"
            f"Balance: {balance:.2f} ETB",
            reply_markup=wallet_keyboard()
        )

        return

    # BALANCE

    if data == "balance":

        balance = get_balance(
            query.from_user.id
        )

        await query.edit_message_text(
            f"💰 YOUR BALANCE\n\n"
            f"{balance:.2f} ETB",
            reply_markup=wallet_keyboard()
        )

        return

    # DEPOSIT

    if data == "deposit":

        await deposit_start(
            query,
            context
        )

        return

    # DEPOSIT METHOD

    if data == "dep_telebirr":

        await deposit_method(
            query,
            context,
            "telebirr"
        )

        return

    if data == "dep_cbe":

        await deposit_method(
            query,
            context,
            "cbe"
        )

        return

    # DEPOSIT AMOUNTS

    if data.startswith("dep_"):

        value = data.replace(
            "dep_",
            "",
            1
        )

        if value == "other":

            context.user_data[
                "waiting_deposit_amount"
            ] = True

            await query.edit_message_text(
                "✏️ OTHER AMOUNT\n\n"
                f"Minimum: {MIN_DEPOSIT} ETB\n\n"
                "የሚፈልጉትን amount ያስገቡ።"
            )

            return

        if value.isdigit():

            await deposit_amount(
                query,
                context,
                float(value)
            )

            return

    # I PAID

    if data == "dep_paid":

        await deposit_paid(
            query,
            context
        )

        return

    # WITHDRAW

    if data == "withdraw":

        await withdraw_start(
            query,
            context
        )

        return

    # WITHDRAW AMOUNTS

    if data.startswith("wd_"):

        value = data.replace(
            "wd_",
            "",
            1
        )

        if value == "other":

            context.user_data[
                "waiting_withdraw_amount"
            ] = True

            await query.edit_message_text(
                "✏️ OTHER WITHDRAW AMOUNT\n\n"
                f"Minimum: {MIN_WITHDRAW} ETB\n\n"
                "የሚፈልጉትን amount ያስገቡ።"
            )

            return

        if value.isdigit():

            await withdraw_amount(
                query,
                context,
                float(value)
            )

            return

    # CONFIRM WITHDRAW

    if data == "wd_confirm":

        await confirm_withdraw(
            query,
            context
        )

        return

    # ADMIN

    if data.startswith(
        "admin_approve_"
    ):

        payment_id = int(
            data.split("_")[-1]
        )

        await admin_review(
            query,
            payment_id,
            True
        )

        return

    if data.startswith(
        "admin_reject_"
    ):

        payment_id = int(
            data.split("_")[-1]
        )

        await admin_review(
            query,
            payment_id,
            False
        )

        return

@app.post("/api/deposit")
def api_deposit(data: dict):

    user_id = int(
        data.get("user_id", 0)
    )

    amount = float(
        data.get("amount", 0)
    )

    method = str(
        data.get("method", "")
    ).lower()

    reference = str(
        data.get("reference", "")
    )

    screenshot = str(
        data.get(
            "screenshot_file_id",
            ""
        )
    )

    ensure_user(user_id)

    if amount < MIN_DEPOSIT:

        raise HTTPException(
            status_code=400,
            detail=f"Minimum deposit is {MIN_DEPOSIT} ETB"
        )

    if method not in (
        "telebirr",
        "cbe"
    ):

        raise HTTPException(
            status_code=400,
            detail="Deposit method must be Telebirr or CBE"
        )

    if len(reference) < 3:

        raise HTTPException(
            status_code=400,
            detail="Transaction reference required"
        )

    payment_id = create_payment(
        user_id,
        "DEPOSIT",
        method,
        amount,
        reference,
        screenshot
    )

    return {
        "ok": True,
        "payment_id": payment_id,
        "status": "PENDING"
    }

@app.post("/api/withdraw")
def api_withdraw(data: dict):

    user_id = int(
        data.get("user_id", 0)
    )

    amount = float(
        data.get("amount", 0)
    )

    account = str(
        data.get("account_number", "")
    )

    name = str(
        data.get("account_name", "")
    )

    ensure_user(user_id)

    if amount < MIN_WITHDRAW:

        raise HTTPException(
            status_code=400,
            detail=f"Minimum withdrawal is {MIN_WITHDRAW} ETB"
        )

    if not account.isdigit():

        raise HTTPException(
            status_code=400,
            detail="Invalid Telebirr account"
        )

    if len(name) < 2:

        raise HTTPException(
            status_code=400,
            detail="Account name required"
        )

    if amount > get_balance(user_id):

        raise HTTPException(
            status_code=400,
            detail="Insufficient balance"
        )

    payment_id = create_payment(
        user_id,
        "WITHDRAWAL",
        "telebirr",
        amount,
        account_number=account,
        account_name=name
    )

    return {
        "ok": True,
        "payment_id": payment_id,
        "status": "PENDING",
        "method": "telebirr"
    }
